这个题是 P7532 的弱化版。参考[这篇题解](https://www.luogu.com.cn/article/xkw9p6p9)。只是性质的证明稍有不同，实际上是等价的。请先阅读上面的题解，本题解中用到了上面的题解中的所有东西（小练习除外）。

实际上，还有更加优雅的转移方法。它不需要二维前缀和，时间空间都更加优秀。

我们有容斥做法！

首先有一步转化。我们把状态 $(l,r)$ 转化为一个二维平面上的点。这样更加直观。

### 状态 $1$

我们注意到这个是上一轮的状态 $1$ 的左上子矩阵和再加 $1$。我们就先二维前缀和然后再给所有（合法的）状态加 $1$ 即可。

### 状态 $2$

这个稍微复杂一点，因为同时涉及到两种状态。对于状态 $2$ 是纯粹的左下子矩阵和，对于状态 $1$ 则要求 $x$ 坐标不能相同。那么，我们将状态 $1$ 右移一格（代码中不需要这一步，只需要变换一下坐标即可），然后相当于把两个状态的平面叠起来（按位加法，代码中也不需要这一步），然后二维前缀和即可。

### 状态 $3$

同状态 $2$。这个是右上子矩阵和，但是状态 $1$ 需要向下移动一格。

### 状态 $4$

虽说是最复杂的一个，但是其实思维含量不大，因为我们已经在上面推出了一般的方法。这个是右下子矩阵和。我们将状态 $1$ 向左上方移动一格（左、上分别一格），状态 $2$ 向左移动一格，状态 $3$ 向上移动一格，状态 $4$ 不移动，再分别叠起来即可。

### 总结

时空复杂度均不变（$O(mn^2)$），代码更加简洁，常数更小，思维含量略高。

#include <cstdio>

using namespace std;

unsigned long long f[155][155][155][2][2];

int main()
{
	int n, m;
	scanf("%d%d", &n, &m);
	unsigned long long sum = 0;
	for(int i=1;i<=n;i++)
	{
		for(int l=m;l>=1;l--)
		{
			for(int r=l;r<=m;r++)
			{
				f[i][l][r][0][0] = (f[i][l+1][r][0][0] + f[i][l][r-1][0][0] - f[i][l+1][r-1][0][0] + f[i-1][l][r][0][0] + 1000000007ull * 5) % 1000000007;
			}
		}
		for(int l=m;l>=1;l--)
		{
			for(int r=l;r<=m;r++)
			{
				f[i][l][r][0][0] = (f[i][l][r][0][0] + 1) % 1000000007;
				sum = (sum + f[i][l][r][0][0]) % 1000000007;
				// printf("f[%d][%d][%d][0][0] = %llu\n", i, l, r, f[i][l][r][0][0]);
			}
		}
		for(int l=1;l<=m;l++)
		{
			for(int r=l;r<=m;r++)
			{
				f[i][l][r][1][0] = (f[i][l-1][r][1][0] + f[i][l][r-1][1][0] - f[i][l-1][r-1][1][0] + f[i-1][l][r][1][0] + f[i-1][l-1][r][0][0] + 1000000007ull * 5) % 1000000007;
				sum = (sum + f[i][l][r][1][0]) % 1000000007;
			}
		}
		for(int l=m;l>=1;l--)
		{
			for(int r=m;r>=l;r--)
			{
				f[i][l][r][0][1] = (f[i][l+1][r][0][1] + f[i][l][r+1][0][1] - f[i][l+1][r+1][0][1] + f[i-1][l][r][0][1] + f[i-1][l][r+1][0][0] + 1000000007ull * 5) % 1000000007;
				sum = (sum + f[i][l][r][0][1]) % 1000000007;
			}
		}
		for(int l=1;l<=m;l++)
		{
			for(int r=m;r>=l;r--)
			{
				f[i][l][r][1][1] = (f[i][l-1][r][1][1] + f[i][l][r+1][1][1] - f[i][l-1][r+1][1][1] + f[i-1][l][r][1][1] + f[i-1][l-1][r+1][0][0] + f[i-1][l][r+1][1][0] + f[i-1][l-1][r][0][1] + 1000000007ull * 5) % 1000000007;
				sum = (sum + f[i][l][r][1][1]) % 1000000007;
			}
		}
	}
	printf("%lld\n", sum);
	return 0;
}
